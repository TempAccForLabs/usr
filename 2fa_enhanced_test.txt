# Signup and get TOTP code in one command - backend 2fa signup test proper

TEMP_TOKEN=$(curl -s -X POST "http://localhost:8000/auth/signup" \
  -H "Content-Type: application/json" \
  -d '{"first_name":"Test","last_name":"User","username":"testuser","email":"test@example.com","password":"test123"}' | \
  jq -r '.temp_token') && \
TOTP_CODE=$(curl -s -X GET "http://localhost:8000/auth/debug/current-totp" -H "Authorization: Bearer $TEMP_TOKEN" | \
  jq -r '.current_code') && \
curl -s -X POST "http://localhost:8000/auth/verify-2fa-setup" \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer $TEMP_TOKEN" \
  -d "{\"code\": \"$TOTP_CODE\"}" | jq

# Enhanced 2fa full login flow proper + debugging token
"""
    Login → get temp token

    Get TOTP code

    Complete 2FA verification → get full access token

    Debug the full token to see if is_2fa_verified: true is set
"""

TEMP_TOKEN=$(curl -s -X POST "http://localhost:8000/auth/login" \
  -H "Content-Type: application/json" \
  -d '{"email":"test@example.com","password":"test123"}' | \
  jq -r '.access_token') && \
TOTP_CODE=$(curl -s -X GET "http://localhost:8000/auth/debug/current-totp" \
  -H "Authorization: Bearer $TEMP_TOKEN" | \
  jq -r '.current_code') && \
FULL_TOKEN=$(curl -s -X POST "http://localhost:8000/auth/verify-2fa" \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer $TEMP_TOKEN" \
  -d "{\"code\": \"$TOTP_CODE\"}" | jq -r '.access_token') && \
curl -s -X GET "http://localhost:8000/auth/debug/token-contents" \
  -H "Authorization: Bearer $FULL_TOKEN" | jq
