uvicorn src:app --reload  --host 0.0.0.0 --port 8000



curl -s -X POST "http://localhost:8000/auth/signup" \
-H "Content-Type: application/json" \
-d '{
  "first_name": "Test",
  "last_name": "User", 
  "username": "testuser",
  "email": "test@example.com",
  "password": "test123"
}' > /dev/null && \
ACCESS_TOKEN=$(curl -s -X POST "http://localhost:8000/auth/login" \
-H "Content-Type: application/json" \
-d '{"email":"test@example.com","password":"test123"}' | jq -r '.access_token') && \
curl -s -X GET "http://localhost:8000/auth/debug/current-totp" \
-H "Authorization: Bearer $ACCESS_TOKEN" | jq -r '.current_code'
